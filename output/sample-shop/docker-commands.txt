#!/usr/bin/env bash
# ============================================================================
#  docker compose up -d と同じ構成を docker コマンドだけで起動する手順
#  (compose2docker 1.0.0 で自動生成)
# ----------------------------------------------------------------------------
#  プロジェクト名 : sample-shop
#  元の compose   : /opt/sample-shop/compose.yaml
#  (変換した場所 : /mnt/c/Users/taka_/Claude/Compose_Docker_Converter/examples/sample-shop/compose.yaml)
#  読み込み方法   : docker compose config (Docker Compose 5.5.1 が解釈した結果)
#  生成日時       : 2026-10-10 15:06:28
#  実行環境       : RHEL 9.8 + Docker Engine (docker-ce 25 以降 / 推奨 29.x)
#
#  ■ 使い方
#   1) RHEL に docker を使えるユーザー (root か docker グループ) でログインする
#   2) STEP 1 から順番に、STEP ごとのかたまりをコピーしてターミナルに貼り付け
#      Enter を押す (結果を確かめてから次の STEP へ進むと安心です)
#      まとめて実行する場合: bash docker-commands.txt
#   3) 片付け (compose down 相当) は docker-commands-down.txt を使う
#   ※ 2 回目以降に実行するときは、先に片付けをしてください (同じ名前があるとエラーになります)
#
#  ■ 変換時の注意
#   ・services.toolbox は profiles (debug) が有効になっていないため起動しま
#   せん (compose の up と同じ)。起動したいときは --profile debug を付けて変
#   換し直してください。
# ============================================================================

# ----------------------------------------------------------------------------
# STEP 1/27 [準備] 作業フォルダへ移動し、Docker が使えるか確認する
#   履歴展開 (!) を止めて貼り付け事故を防ぎ、プロジェクトのフォルダへ移動し
#   て、docker コマンドとサーバー (dockerd) の両方が動いているかを確かめます。
# ----------------------------------------------------------------------------
set +H
cd /opt/sample-shop
docker version

# ----------------------------------------------------------------------------
# STEP 2/27 [ネットワーク作成] ネットワーク「sample-shop_frontend」を作る
#   (compose の対応箇所: networks.frontend)
#   仮想ネットワーク「sample-shop_frontend」を作ります。つながるコンテナ:
#   app、web。同じネットワークのコンテナどうしは、サービス名で呼び合えます。
# ----------------------------------------------------------------------------
docker network create sample-shop_frontend

# ----------------------------------------------------------------------------
# STEP 3/27 [ネットワーク作成] ネットワーク「sample-shop_backend」を作る
#   (compose の対応箇所: networks.backend)
#   仮想ネットワーク「sample-shop_backend」を作ります。つながるコンテナ: db、
#   cache、migrate、app。同じネットワークのコンテナどうしは、サービス名で呼
#   び合えます。
# ----------------------------------------------------------------------------
docker network create --internal sample-shop_backend

# ----------------------------------------------------------------------------
# STEP 4/27 [ボリューム作成] ボリューム「sample-shop_db-data」を作る
#   (compose の対応箇所: volumes.db-data)
#   データを保存しておく名前付きボリューム「sample-shop_db-data」を作ります。
#   使うサービス: db。
# ----------------------------------------------------------------------------
docker volume create sample-shop_db-data

# ----------------------------------------------------------------------------
# STEP 5/27 [ボリューム作成] ボリューム「sample-shop_cache-data」を作る
#   (compose の対応箇所: volumes.cache-data)
#   データを保存しておく名前付きボリューム「sample-shop_cache-data」を作りま
#   す。使うサービス: cache。
# ----------------------------------------------------------------------------
docker volume create sample-shop_cache-data

# ----------------------------------------------------------------------------
# STEP 6/27 [ボリューム作成] ボリューム「sample-shop_app-cache」を作る
#   (compose の対応箇所: volumes.app-cache)
#   データを保存しておく名前付きボリューム「sample-shop_app-cache」を作りま
#   す。使うサービス: app。
# ----------------------------------------------------------------------------
docker volume create sample-shop_app-cache

# ----------------------------------------------------------------------------
# STEP 7/27 [イメージ取得] イメージ「postgres:16-alpine」を用意する
#   (compose の対応箇所: services.db.image)
#   イメージ「postgres:16-alpine」が手元に無いときだけダウンロードします
#   (compose の既定 pull_policy: missing と同じ動き)。
# ----------------------------------------------------------------------------
docker image inspect postgres:16-alpine > /dev/null 2>&1 || docker pull postgres:16-alpine

# ----------------------------------------------------------------------------
# STEP 8/27 [イメージ取得] イメージ「valkey/valkey:8-alpine」を用意する
#   (compose の対応箇所: services.cache.image)
#   イメージ「valkey/valkey:8-alpine」が手元に無いときだけダウンロードします
#   (compose の既定 pull_policy: missing と同じ動き)。
# ----------------------------------------------------------------------------
docker image inspect valkey/valkey:8-alpine > /dev/null 2>&1 || docker pull valkey/valkey:8-alpine

# ----------------------------------------------------------------------------
# STEP 9/27 [イメージ取得] イメージ「nginx:1.27-alpine」を用意する
#   (compose の対応箇所: services.web.image)
#   イメージ「nginx:1.27-alpine」が手元に無いときだけダウンロードします
#   (compose の既定 pull_policy: missing と同じ動き)。
# ----------------------------------------------------------------------------
docker image inspect nginx:1.27-alpine > /dev/null 2>&1 || docker pull nginx:1.27-alpine

# ----------------------------------------------------------------------------
# STEP 10/27 [イメージ取得] イメージ「busybox:1.36」を用意する
#   (compose の対応箇所: services.app.pre_start)
#   イメージ「busybox:1.36」が手元に無いときだけダウンロードします (compose
#   の既定 pull_policy: missing と同じ動き)。
# ----------------------------------------------------------------------------
docker image inspect busybox:1.36 > /dev/null 2>&1 || docker pull busybox:1.36

# ----------------------------------------------------------------------------
# STEP 11/27 [イメージ作成] サービス「app」のイメージ「sample-shop/app:1.0.0」をビルドする
#   (compose の対応箇所: services.app.build)
#   Dockerfile (設計図) とビルドコンテキスト (材料フォルダ) からイメージ「
#   sample-shop/app:1.0.0」を作ります。
# ----------------------------------------------------------------------------
docker build \
  --target runtime \
  --build-arg APP_VERSION=1.0.0 \
  --build-arg PYTHON_VERSION=3.12 \
  --label com.example.built-by=compose2docker-sample \
  --tag sample-shop/app:1.0.0 \
  /opt/sample-shop/app

# ----------------------------------------------------------------------------
# STEP 12/27 [ヘルスチェック入りイメージ作成] サービス「cache」用に、ヘルスチェックを焼き込んだイメージ「sample-shop-cache:with-healthcheck」を作る
#   (compose の対応箇所: services.cache.healthcheck)
#   compose の healthcheck は exec 形式 (test: ["CMD", ...]) ですが、docker
#   run の --health-cmd はシェル形式 (/bin/sh -c で実行) しか書けません。同
#   じ動きにするため、元のイメージに HEALTHCHECK だけを追加した派生イメージ
#   を作ります。
# ----------------------------------------------------------------------------
docker build --tag sample-shop-cache:with-healthcheck - <<'C2D_DOCKERFILE'
FROM valkey/valkey:8-alpine
HEALTHCHECK --interval=5s --timeout=3s --retries=10 CMD ["valkey-cli", "ping"]
C2D_DOCKERFILE

# ----------------------------------------------------------------------------
# STEP 13/27 [ヘルスチェック入りイメージ作成] サービス「web」用に、ヘルスチェックを焼き込んだイメージ「sample-shop-web:with-healthcheck」を作る
#   (compose の対応箇所: services.web.healthcheck)
#   compose の healthcheck は exec 形式 (test: ["CMD", ...]) ですが、docker
#   run の --health-cmd はシェル形式 (/bin/sh -c で実行) しか書けません。同
#   じ動きにするため、元のイメージに HEALTHCHECK だけを追加した派生イメージ
#   を作ります。
# ----------------------------------------------------------------------------
docker build --tag sample-shop-web:with-healthcheck - <<'C2D_DOCKERFILE'
FROM nginx:1.27-alpine
HEALTHCHECK --interval=10s --timeout=3s --retries=5 CMD ["wget", "-q", "-O", "/dev/null", "http://127.0.0.1/healthz"]
C2D_DOCKERFILE

# ----------------------------------------------------------------------------
# STEP 14/27 [コンテナ起動] サービス「db」のコンテナ「sample-shop-db-1」を起動する
#   (compose の対応箇所: services.db)
#   イメージ「postgres:16-alpine」からコンテナ「sample-shop-db-1」を作って起
#   動します (サービス「db」)。
# ----------------------------------------------------------------------------
docker run \
  --detach \
  --name sample-shop-db-1 \
  --restart unless-stopped \
  --network sample-shop_backend \
  --network-alias db \
  --volume sample-shop_db-data:/var/lib/postgresql/data \
  --volume /opt/sample-shop/db/init.sql:/docker-entrypoint-initdb.d/10-init.sql:ro \
  --mount type=bind,source=/opt/sample-shop/secrets/db_password.txt,target=/run/secrets/db_password,readonly \
  --env POSTGRES_DB=shop \
  --env POSTGRES_PASSWORD_FILE=/run/secrets/db_password \
  --env POSTGRES_USER=shop \
  --shm-size 128m \
  --log-driver json-file \
  --log-opt max-file=3 \
  --log-opt max-size=10m \
  --health-cmd 'pg_isready -U shop -d shop' \
  --health-interval 5s \
  --health-timeout 3s \
  --health-start-period 10s \
  --health-retries 10 \
  --stop-timeout 30 \
  postgres:16-alpine

# ----------------------------------------------------------------------------
# STEP 15/27 [コンテナ起動] サービス「cache」のコンテナ「sample-shop-cache-1」を起動する
#   (compose の対応箇所: services.cache)
#   イメージ「sample-shop-cache:with-healthcheck」からコンテナ「
#   sample-shop-cache-1」を作って起動します (サービス「cache」)。
# ----------------------------------------------------------------------------
docker run \
  --detach \
  --name sample-shop-cache-1 \
  --restart unless-stopped \
  --network sample-shop_backend \
  --network-alias cache \
  --volume sample-shop_cache-data:/data \
  --log-driver json-file \
  --log-opt max-file=3 \
  --log-opt max-size=10m \
  sample-shop-cache:with-healthcheck \
  valkey-server --appendonly yes --save ''

# ----------------------------------------------------------------------------
# STEP 16/27 [起動待ち] sample-shop-db-1 が healthy になるまで待つ
#   (compose の対応箇所: services.migrate.depends_on)
#   サービス「migrate」は depends_on で db が「service_healthy (ヘルスチェッ
#   クに合格)」になるのを待ってから起動する設定です。同じように、合格するま
#   で 2 秒おきに状態を確認して待ちます。
#   ※ NG と表示されたら、ここで手を止めて docker logs <コンテナ名> で原因を
#   確認してください (この後のコマンドを続けても、依存先が正常でないため動か
#   ない可能性があります)。
# ----------------------------------------------------------------------------
for C in sample-shop-db-1; do
  echo "[待機] ${C} が healthy (正常) になるまで待ちます..."
  while :; do
    S="$(docker inspect --format '{{.State.Status}}:{{if .State.Health}}{{.State.Health.Status}}{{end}}' "${C}" 2>&1)"
    case "${S}" in
      running:healthy) echo "[OK] ${C} は healthy です"; break ;;
      running:starting|restarting:*) sleep 2 ;;
      *) echo "[NG] ${C} の状態は ${S} です。docker logs ${C} で原因を確認してください"; break ;;
    esac
  done
done

# ----------------------------------------------------------------------------
# STEP 17/27 [コンテナ起動] サービス「migrate」のコンテナ「sample-shop-migrate-1」を起動する
#   (compose の対応箇所: services.migrate)
#   イメージ「postgres:16-alpine」からコンテナ「sample-shop-migrate-1」を作
#   って起動します (サービス「migrate」)。
# ----------------------------------------------------------------------------
docker run \
  --detach \
  --name sample-shop-migrate-1 \
  --network sample-shop_backend \
  --network-alias migrate \
  --volume /opt/sample-shop/db/migrations:/migrations:ro \
  --mount type=bind,source=/opt/sample-shop/secrets/db_password.txt,target=/run/secrets/db_password,readonly \
  --log-driver json-file \
  --log-opt max-file=3 \
  --log-opt max-size=10m \
  postgres:16-alpine \
  sh \
  -c \
  $'export PGPASSWORD="$(cat /run/secrets/db_password)"\npsql -h db -U shop -d shop -v ON_ERROR_STOP=1 -f /migrations/001_items.sql\necho "migration done"\n'

# ----------------------------------------------------------------------------
# STEP 18/27 [起動待ち] sample-shop-cache-1・sample-shop-db-1 が healthy になるまで待つ
#   (compose の対応箇所: services.app.depends_on)
#   サービス「app」は depends_on で cache・db が「service_healthy (ヘルスチ
#   ェックに合格)」になるのを待ってから起動する設定です。同じように、合格す
#   るまで 2 秒おきに状態を確認して待ちます。
#   ※ NG と表示されたら、ここで手を止めて docker logs <コンテナ名> で原因を
#   確認してください (この後のコマンドを続けても、依存先が正常でないため動か
#   ない可能性があります)。
# ----------------------------------------------------------------------------
for C in sample-shop-cache-1 sample-shop-db-1; do
  echo "[待機] ${C} が healthy (正常) になるまで待ちます..."
  while :; do
    S="$(docker inspect --format '{{.State.Status}}:{{if .State.Health}}{{.State.Health.Status}}{{end}}' "${C}" 2>&1)"
    case "${S}" in
      running:healthy) echo "[OK] ${C} は healthy です"; break ;;
      running:starting|restarting:*) sleep 2 ;;
      *) echo "[NG] ${C} の状態は ${S} です。docker logs ${C} で原因を確認してください"; break ;;
    esac
  done
done

# ----------------------------------------------------------------------------
# STEP 19/27 [完了待ち] sample-shop-migrate-1 が正常終了するまで待つ
#   (compose の対応箇所: services.app.depends_on)
#   サービス「app」は depends_on で「service_completed_successfully (正常に
#   終了)」を待つ設定です。docker wait で終了を待ち、終了コードが 0 (成功)
#   かを確かめます。
# ----------------------------------------------------------------------------
echo "[待機] sample-shop-migrate-1 が終わるまで待ちます..."
EXIT_CODE="$(docker wait sample-shop-migrate-1)"
echo "sample-shop-migrate-1 の終了コード: ${EXIT_CODE} (0 なら成功)"

# ----------------------------------------------------------------------------
# STEP 20/27 [コンテナ作成] サービス「app」のコンテナ「sample-shop-app-1」を作る (まだ起動しない)
#   (compose の対応箇所: services.app)
#   コンテナ「sample-shop-app-1」をイメージ「sample-shop/app:1.0.0」から作り
#   ますが、まだ起動しません。起動前にファイルの投入や初期化 (pre_start) を
#   行うためです。
#   ※ 複数のネットワークを docker run 1 回でつなぐ書き方 (--network
#   name=...,alias=...) は Docker Engine 25 以降で使えます。RHEL 9.8 で最新
#   の docker-ce (29.x) を入れていれば問題ありません。
# ----------------------------------------------------------------------------
docker create \
  --name sample-shop-app-1 \
  --restart unless-stopped \
  --init \
  --user 10001:10001 \
  --network name=sample-shop_backend,alias=app \
  --network name=sample-shop_frontend,alias=app,alias=api \
  --add-host host.docker.internal:host-gateway \
  --volume sample-shop_app-cache:/var/cache/shop \
  --tmpfs /tmp:size=16m \
  --env-file /opt/sample-shop/env/app.env \
  --env API_TOKEN=s3cr3t-token \
  --env CACHE_HOST=cache \
  --env DB_HOST=db \
  --env DB_PORT=5432 \
  --env 'GREETING=Hello from version 1.0.0' \
  --env LOG_LEVEL=info \
  --env SETTINGS_FILE=/etc/shop/settings.ini \
  --ulimit nofile=4096:8192 \
  --cpus 0.5 \
  --memory 256m \
  --log-driver json-file \
  --log-opt max-file=3 \
  --log-opt max-size=10m \
  --health-cmd $'python -c "import urllib.request; urllib.request.urlopen(\'http://127.0.0.1:8000/healthz\', timeout=3)"' \
  --health-interval 5s \
  --health-timeout 3s \
  --health-start-period 5s \
  --health-retries 10 \
  sample-shop/app:1.0.0

# ----------------------------------------------------------------------------
# STEP 21/27 [ファイル投入] 設定ファイル (configs)「app_settings」を sample-shop-app-1 の /etc/shop/settings.ini に入れる
#   (compose の対応箇所: configs.app_settings)
#   compose は 設定ファイル (configs) の中身 (content / environment) を、コ
#   ンテナを起動する前にファイルとして書き込みます。同じように、一時ファイル
#   を作って tar でまとめ、docker cp でコンテナの /etc/shop/settings.ini に
#   入れます。
# ----------------------------------------------------------------------------
mkdir -p /tmp/sample-shop-compose2docker/sample-shop-app-1/etc/shop
cat > /tmp/sample-shop-compose2docker/sample-shop-app-1/etc/shop/settings.ini <<'C2D_CONTENT'
[shop]
currency = JPY
tax_rate = 0.10
C2D_CONTENT
tar --create --owner=0 --group=0 --mode=0444 --directory /tmp/sample-shop-compose2docker/sample-shop-app-1 --file - etc/shop/settings.ini | docker cp - sample-shop-app-1:/
rm -rf /tmp/sample-shop-compose2docker/sample-shop-app-1

# ----------------------------------------------------------------------------
# STEP 22/27 [初期化コンテナ] サービス「app」の起動前の初期化 1 (pre_start) を実行する
#   (compose の対応箇所: services.app.pre_start[0])
#   compose v5.3 で追加された init コンテナ (pre_start) を再現します。本番コ
#   ンテナを作った後・起動する前に、使い捨てのコンテナで準備作業を 1 回だけ
#   行います。
# ----------------------------------------------------------------------------
docker run \
  --rm \
  --volumes-from sample-shop-app-1 \
  --network name=sample-shop_backend,alias=app \
  --network name=sample-shop_frontend,alias=app,alias=api \
  --user root \
  --env-file /opt/sample-shop/env/app.env \
  --env API_TOKEN=s3cr3t-token \
  --env CACHE_HOST=cache \
  --env DB_HOST=db \
  --env DB_PORT=5432 \
  --env 'GREETING=Hello from version 1.0.0' \
  --env LOG_LEVEL=info \
  --env SETTINGS_FILE=/etc/shop/settings.ini \
  busybox:1.36 \
  sh -c 'echo warmed-at-start > /var/cache/shop/warm.txt && chown -R 10001:10001 /var/cache/shop'

# ----------------------------------------------------------------------------
# STEP 23/27 [コンテナ起動] コンテナ「sample-shop-app-1」を起動する
#   (compose の対応箇所: services.app)
#   ファイルの投入・初期化コンテナ (pre_start) が終わったので、コンテナ「
#   sample-shop-app-1」を起動します。
# ----------------------------------------------------------------------------
docker start sample-shop-app-1

# ----------------------------------------------------------------------------
# STEP 24/27 [起動後フック] コンテナ「sample-shop-app-1」で post_start フック 1 を実行する
#   (compose の対応箇所: services.app.post_start[0])
#   compose の post_start フック (コンテナ起動直後に中で実行するコマンド) を
#   docker exec で再現します。
# ----------------------------------------------------------------------------
docker exec sample-shop-app-1 sh -c 'date > /tmp/started-at'

# ----------------------------------------------------------------------------
# STEP 25/27 [起動待ち] sample-shop-app-1 が healthy になるまで待つ
#   (compose の対応箇所: services.web.depends_on)
#   サービス「web」は depends_on で app が「service_healthy (ヘルスチェック
#   に合格)」になるのを待ってから起動する設定です。同じように、合格するまで
#   2 秒おきに状態を確認して待ちます。
#   ※ NG と表示されたら、ここで手を止めて docker logs <コンテナ名> で原因を
#   確認してください (この後のコマンドを続けても、依存先が正常でないため動か
#   ない可能性があります)。
# ----------------------------------------------------------------------------
for C in sample-shop-app-1; do
  echo "[待機] ${C} が healthy (正常) になるまで待ちます..."
  while :; do
    S="$(docker inspect --format '{{.State.Status}}:{{if .State.Health}}{{.State.Health.Status}}{{end}}' "${C}" 2>&1)"
    case "${S}" in
      running:healthy) echo "[OK] ${C} は healthy です"; break ;;
      running:starting|restarting:*) sleep 2 ;;
      *) echo "[NG] ${C} の状態は ${S} です。docker logs ${C} で原因を確認してください"; break ;;
    esac
  done
done

# ----------------------------------------------------------------------------
# STEP 26/27 [コンテナ起動] サービス「web」のコンテナ「sample-shop-web-1」を起動する
#   (compose の対応箇所: services.web)
#   イメージ「sample-shop-web:with-healthcheck」からコンテナ「
#   sample-shop-web-1」を作って起動します (サービス「web」)。
# ----------------------------------------------------------------------------
docker run \
  --detach \
  --name sample-shop-web-1 \
  --restart always \
  --network sample-shop_frontend \
  --network-alias web \
  --publish 8088:80 \
  --volume /opt/sample-shop/web/nginx.conf:/etc/nginx/conf.d/default.conf:ro \
  --tmpfs /var/cache/nginx \
  --tmpfs /var/run \
  --tmpfs /tmp \
  --label com.example.role=frontend \
  --read-only \
  --cap-drop NET_RAW \
  --security-opt no-new-privileges:true \
  --log-driver json-file \
  --log-opt max-file=3 \
  --log-opt max-size=10m \
  --stop-signal SIGQUIT \
  --stop-timeout 15 \
  sample-shop-web:with-healthcheck

# ----------------------------------------------------------------------------
# STEP 27/27 [状態確認] 起動したコンテナの状態を確認する
#   作ったコンテナ (sample-shop-db-1、sample-shop-cache-1、
#   sample-shop-migrate-1、sample-shop-app-1、sample-shop-web-1) が Up (起動
#   中) になっているか、ヘルスチェックの結果 (healthy など) を確認します。
# ----------------------------------------------------------------------------
docker ps \
  --all \
  --format 'table {{.Names}}\t{{.Image}}\t{{.Status}}\t{{.Ports}}'

# ============================================================================
#  ここまでで起動は完了です。片付けは docker-commands-down.txt を参照してください。
# ============================================================================
